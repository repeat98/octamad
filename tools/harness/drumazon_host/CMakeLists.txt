cmake_minimum_required(VERSION 3.22)
project(drumazon_host VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT JUCE_SOURCE)
  message(FATAL_ERROR "Pass -DJUCE_SOURCE=/path/to/JUCE")
endif()
add_subdirectory(${JUCE_SOURCE} juce)
juce_add_console_app(drumazon_host PRODUCT_NAME "Drumazon Reference Host")
target_sources(drumazon_host PRIVATE main.cpp)
target_compile_definitions(drumazon_host PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(drumazon_host PRIVATE juce::juce_audio_processors juce::juce_audio_formats)
